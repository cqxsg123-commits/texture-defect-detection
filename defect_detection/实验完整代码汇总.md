```python
"""通用工具：随机种子、设备选择、热力图可视化。"""

from __future__ import annotations

import random
from pathlib import Path

import cv2
import matplotlib

matplotlib.use("Agg")  # 无界面后端，便于服务器/脚本运行
import matplotlib.pyplot as plt
import numpy as np
import torch

# 中文字体：热力图标题含中文，默认 DejaVu Sans 缺 CJK 字形会渲染成方框
plt.rcParams["font.sans-serif"] = ["Microsoft YaHei", "SimHei", "sans-serif"]
plt.rcParams["axes.unicode_minus"] = False

# ImageNet 均值 / 标准差（与计划书预处理流水线一致）
IMAGENET_MEAN = np.array([0.485, 0.456, 0.406], dtype=np.float32)
IMAGENET_STD = np.array([0.229, 0.224, 0.225], dtype=np.float32)


def set_seed(seed: int = 42) -> None:
    """固定随机种子，保证实验可复现。"""
    random.seed(seed)
    np.random.seed(seed)
    torch.manual_seed(seed)
    torch.cuda.manual_seed_all(seed)
    torch.backends.cudnn.deterministic = True
    torch.backends.cudnn.benchmark = False


def get_device() -> torch.device:
    """返回可用设备：优先 CUDA，否则 CPU。"""
    return torch.device("cuda" if torch.cuda.is_available() else "cpu")


def denormalize(img: np.ndarray) -> np.ndarray:
    """反归一化：ImageNet 归一化图像 -> [0,1] 原始图像 (H,W,3)。"""
    img = img * IMAGENET_STD.reshape(1, 1, 3) + IMAGENET_MEAN.reshape(1, 1, 3)
    return np.clip(img, 0.0, 1.0)


def min_max_norm(x: np.ndarray) -> np.ndarray:
    """逐张图 min-max 归一化到 [0,1]。"""
    x_min, x_max = x.min(), x.max()
    if x_max - x_min < 1e-8:
        return np.zeros_like(x)
    return (x - x_min) / (x_max - x_min)


def smooth_map(anomaly_map: np.ndarray, sigma: int = 4) -> np.ndarray:
    """高斯平滑异常热力图，去除散点噪声。"""
    kernel = int(sigma * 4) | 1  # 保证奇数
    return cv2.GaussianBlur(anomaly_map, (kernel, kernel), sigma)


def resize_map(anomaly_map: np.ndarray, size: tuple[int, int]) -> np.ndarray:
    """双线性放大异常图到目标尺寸 (H,W)。"""
    return cv2.resize(anomaly_map, (size[1], size[0]), interpolation=cv2.INTER_LINEAR)


def save_triplet(
    save_path: Path,
    image: np.ndarray,
    anomaly_map: np.ndarray,
    mask: np.ndarray | None = None,
    title: str = "",
) -> None:
    """保存「原图 + 热力图 + 缺陷掩码」三联图（报告素材）。

    Args:
        save_path: 输出文件路径。
        image: 原始图像 (H,W,3)，范围 [0,1]。
        anomaly_map: 异常热力图 (H,W)，任意范围（内部自动 min-max 归一化）。
        mask: 真实缺陷掩码 (H,W)，0/1；为 None 时只画两联。
        title: 图片标题（含图像级得分等信息）。
    """
    n_cols = 3 if mask is not None else 2
    fig, axes = plt.subplots(1, n_cols, figsize=(4.5 * n_cols, 4.2))
    # n_cols 恒 >= 2，subplots 返回的 axes 本身就是一维数组，直接使用
    axes = np.atleast_1d(axes)

    axes[0].imshow(image)
    axes[0].set_title("原图")
    im = axes[1].imshow(min_max_norm(anomaly_map), cmap="jet")
    axes[1].set_title("异常热力图")
    if mask is not None:
        axes[2].imshow(mask, cmap="gray")
        axes[2].set_title("真实缺陷掩码")

    for ax in axes:
        ax.axis("off")
    if title:
        fig.suptitle(title, fontsize=11)
    plt.colorbar(im, ax=axes[1], fraction=0.046, pad=0.04)
    fig.tight_layout()
    fig.savefig(save_path, dpi=120, bbox_inches="tight")
    plt.close(fig)


def heatmap_overlay(image: np.ndarray, anomaly_map: np.ndarray, alpha: float = 0.55) -> np.ndarray:
    """将热力图以半透明方式叠加到原图上 (H,W,3) uint8，用于 Gradio 展示。"""
    heat = cv2.applyColorMap(
        (min_max_norm(anomaly_map) * 255).astype(np.uint8), cv2.COLORMAP_JET
    )
    base = (image * 255).astype(np.uint8)[..., ::-1]  # RGB -> BGR
    return cv2.addWeighted(heat, alpha, base, 1 - alpha, 0)
```

### 0.4 文件：`src/metrics.py`（评估指标）

```python
"""评估指标：图像级 AUROC、像素级 AUROC（计划书 2. 任务定义）。

协议（与 Anomalib 官方一致，保证 3 个模型对比公平）：
- 图像级异常得分 = 原始（未归一化）异常热力图的最大值
- 像素级 AUROC：每张异常图先做逐图 min-max 归一化，再全局展平计算
"""

from __future__ import annotations

import numpy as np
from sklearn.metrics import roc_auc_score


def compute_image_auroc(scores: np.ndarray, labels: np.ndarray) -> float:
    """图像级 AUROC。

    Args:
        scores: 每张测试图的异常得分 (N,)，越大越异常。
        labels: 真实标签 (N,)，0=正常 1=缺陷。
    """
    if len(np.unique(labels)) < 2:
        return float("nan")
    return float(roc_auc_score(labels, scores))


def compute_pixel_auroc(anomaly_maps: np.ndarray, masks: np.ndarray) -> float:
    """像素级 AUROC。

    Args:
        anomaly_maps: 原始异常热力图 (N,H,W)，任意范围。
        masks: 真实缺陷掩码 (N,H,W)，0/1。
    """
    maps = anomaly_maps.reshape(len(anomaly_maps), -1)
    masks = masks.reshape(len(masks), -1)

    # 逐图 min-max 归一化（异常图最大值位置不固定，需先归一化再跨图比较）
    normalized = []
    for m in maps:
        m_min, m_max = m.min(), m.max()
        if m_max - m_min < 1e-8:
            normalized.append(np.zeros_like(m))
        else:
            normalized.append((m - m_min) / (m_max - m_min))

    flat_maps = np.concatenate(normalized).ravel()
    flat_masks = masks.ravel()
    return float(roc_auc_score(flat_masks, flat_maps))


def anomaly_map_to_score(anomaly_map: np.ndarray) -> float:
    """原始异常图 -> 图像级异常得分（取最大值，Anomalib 协议）。"""
    return float(anomaly_map.max())
```

### 0.5 文件：`src/evaluation.py`（统一评估流程）

```python
"""统一评估流程：图像级 AUROC、PixelAUROC、单张推理耗时、对比表与热力图样例。

3 个模型共用本模块，保证实验条件严格一致（计划书 4.3 对比实验设置）。
"""

from __future__ import annotations

import os
import time
from pathlib import Path

import numpy as np
import pandas as pd
import torch

from src.metrics import compute_image_auroc, compute_pixel_auroc
from src.models.base import BaseAnomalyModel
from src.models.padim_anomalib import PadimAnomalibModel
from src.models.patchcore import PatchCoreModel
from src.utils import denormalize, save_triplet

TIMING_RUNS = 20  # 推理耗时测量的重复次数


def _norm_path(p) -> str:
    return os.path.normpath(os.path.abspath(str(p)))


def evaluate_model(
    model: BaseAnomalyModel,
    test_loader,
    heatmap_dir: str | Path | None = None,
    n_heatmap_samples: int = 3,
) -> dict:
    """评估单个模型。

    Returns:
        {"model", "image_auroc", "pixel_auroc", "mean_inference_ms", "n_test"}
    """
    print(f"[eval] 评估模型：{model.name}")
    ds = test_loader.dataset

    # ---- 收集全部测试图的原始异常热力图（保持数据集顺序） ----
    if isinstance(model, PadimAnomalibModel):
        # Anomalib 适配层：整测试集预测后按路径对齐
        maps_by_path = {_norm_path(k): v for k, v in model.predict_dataset().items()}
        raw_maps = np.stack([maps_by_path[_norm_path(ds.image_paths[i])] for i in range(len(ds))])
    elif isinstance(model, PatchCoreModel):
        # 官方评估协议：FAISS CPU 检索生成热力图（指标与优化前完全一致；
        # GPU 批量检索仅用于部署与耗时测试，两者地图差异为 fp32 求和次序噪声）
        maps = []
        for batch in test_loader:
            for i in range(batch["image"].shape[0]):
                maps.append(model.predict_image_faiss(batch["image"][i : i + 1]))
        raw_maps = np.stack(maps)
    else:
        maps = []
        model_predict = model.predict
        for batch in test_loader:
            maps.append(model_predict(batch["image"].to(model.device)))
        raw_maps = np.concatenate(maps, axis=0)

    labels = np.array(ds.labels)
    masks = np.stack([np.asarray(ds[i]["mask"][0]) for i in range(len(ds))])

    # ---- 指标 ----
    scores = raw_maps.max(axis=(1, 2))  # 图像级得分 = 原始热力图最大值
    image_auroc = compute_image_auroc(scores, labels)
    pixel_auroc = compute_pixel_auroc(raw_maps, masks)

    # ---- 推理耗时 ----
    mean_ms = _measure_time(model)

    result = {
        "model": model.name,
        "image_auroc": round(image_auroc, 4),
        "pixel_auroc": round(pixel_auroc, 4),
        "mean_inference_ms": round(mean_ms, 2),
        "n_test": len(ds),
    }
    print(
        f"[eval] {model.name}: Image AUROC={result['image_auroc']:.4f} | "
        f"Pixel AUROC={result['pixel_auroc']:.4f} | 推理耗时 {result['mean_inference_ms']:.2f} ms/张"
    )

    # ---- 热力图样例（报告素材） ----
    if heatmap_dir is not None:
        heatmap_dir = Path(heatmap_dir) / model.name
        heatmap_dir.mkdir(parents=True, exist_ok=True)
        defect_idx = np.where(labels == 1)[0]
        normal_idx = np.where(labels == 0)[0]
        for tag, idxs in (("defect", defect_idx), ("normal", normal_idx)):
            if len(idxs) == 0:
                continue
            for k in range(min(n_heatmap_samples, len(idxs))):
                i = idxs[k]
                img = denormalize(ds[i]["image"].numpy().transpose(1, 2, 0))
                save_triplet(
                    heatmap_dir / f"{tag}_{k}.png",
                    img,
                    raw_maps[i],
                    np.asarray(ds[i]["mask"][0]) if labels[i] == 1 else None,
                    title=f"{model.name} | score={scores[i]:.4f} | label={labels[i]}",
                )
        print(f"[eval] 热力图样例已保存 -> {heatmap_dir}")

    return result


def _measure_time(model: BaseAnomalyModel) -> float:
    """测量单张推理耗时（ms）：单图 predict_image 重复 20 次取均值（含 GPU 同步）。

    三模型统一口径，测的是实际部署的单张推理路径：
    - PatchCore：批量前向提取 + GPU 分块 GEMM 记忆库检索（predict_image）
    - PaDiM：直接前向 + GPU 马氏距离（绕开 Engine 调度，predict_image）
    - CutPaste：单次前向 + GradCAM
    """
    device = model.device
    sample = torch.randn(1, 3, model.image_size, model.image_size).to(device)
    model.predict_image(sample)  # 预热（含首次权重搬移）
    if device.type == "cuda":
        torch.cuda.synchronize()
    t0 = time.perf_counter()
    for _ in range(TIMING_RUNS):
        model.predict_image(sample)
    if device.type == "cuda":
        torch.cuda.synchronize()
    return (time.perf_counter() - t0) / TIMING_RUNS * 1000.0


def make_comparison_table(results: list[dict], save_dir: str | Path) -> pd.DataFrame:
    """生成对比表：保存 CSV + Markdown，返回 DataFrame。"""
    df = pd.DataFrame(results).sort_values("pixel_auroc", ascending=False)
    save_dir = Path(save_dir)
    save_dir.mkdir(parents=True, exist_ok=True)
    df.to_csv(save_dir / "comparison.csv", index=False, encoding="utf-8-sig")
    with open(save_dir / "comparison.md", "w", encoding="utf-8") as f:
        f.write(df.to_markdown(index=False))
    print("[eval] 对比表已保存 ->", save_dir)
    return df
```

```python
"""实验脚本公共工具：配置加载、路径解析、模型训练/加载逻辑。"""

from __future__ import annotations

from pathlib import Path

import torch
import yaml

from src import CHECKPOINT_DIR, CONFIG_DIR, PROJECT_ROOT, RESULTS_DIR


def load_config(path: str | Path | None = None) -> dict:
    """加载 YAML 实验配置。"""
    path = Path(path) if path else CONFIG_DIR / "experiment.yaml"
    with open(path, encoding="utf-8") as f:
        return yaml.safe_load(f)


def resolve(root: str | Path) -> Path:
    """相对路径 -> 项目根绝对路径。"""
    root = Path(root)
    return root if root.is_absolute() else PROJECT_ROOT / root


def get_or_train(model, checkpoint_name: str | None, train_loader, retrain: bool = False):
    """已有权重则直接加载，否则训练并保存（--retrain 强制重训）。"""
    if checkpoint_name and not retrain:
        ckpt = resolve(CHECKPOINT_DIR) / checkpoint_name
        if ckpt.exists():
            model.load(ckpt)
            return model
    model.train(train_loader)
    if checkpoint_name:
        model.save(resolve(CHECKPOINT_DIR) / checkpoint_name)
    return model


def release_memory(*objs) -> None:
    """释放模型显存（计划书风险应对：串行训练、及时释放）。"""
    for o in objs:
        del o
    if torch.cuda.is_available():
        torch.cuda.empty_cache()
```

### 0.8 文件：`src/data/dataset.py`（数据集模块）

```python
"""自定义 MVTec AD 数据集模块（计划书 4.4 节：自定义数据集模块）。

- 训练集：仅 metal_nut/train/good 正常样本
- 测试集：metal_nut/test 下 good + 全部缺陷类型，缺陷样本附带 ground_truth 掩码
- 预处理：统一 RGB、resize 256×256、ImageNet 均值方差归一化
"""

from __future__ import annotations

from pathlib import Path

import cv2
import numpy as np
import torch
from torch.utils.data import DataLoader, Dataset

from .augmentations import get_test_transform, get_train_transform

# metal_nut 的四类缺陷（MVTec AD 官方类别）
DEFECT_TYPES = ["bent", "color", "flip", "scratch"]


class MVTecDataset(Dataset):
    """MVTec AD metal_nut 数据集。

    Args:
        root: mvtec_anomaly_detection 根目录（其下为 metal_nut/）。
        category: 类别名，默认 metal_nut。
        split: "train"（仅正常样本）或 "test"（正常 + 缺陷，含掩码）。
        image_size: 输入图像尺寸。
        use_augmentation: 训练集是否启用数据增强（实验 3 开关）。
    """

    def __init__(
        self,
        root,
        category: str = "metal_nut",
        split: str = "train",
        image_size: int = 256,
        use_augmentation: bool = True,
    ):
        super().__init__()
        assert split in ("train", "test"), f"split 必须是 train/test，收到 {split}"
        self.root = Path(root)
        self.category = category
        self.split = split
        self.image_size = image_size
        self.transform = (
            get_train_transform(use_augmentation) if split == "train" else get_test_transform()
        )

        self.image_paths: list[str] = []
        self.mask_paths: list[str | None] = []
        self.labels: list[int] = []  # 0=正常 1=缺陷

        base = self.root / category
        if split == "train":
            # 训练集只有正常样本
            good_dir = base / "train" / "good"
            for p in sorted(good_dir.glob("*.png")):
                self.image_paths.append(str(p))
                self.mask_paths.append(None)
                self.labels.append(0)
        else:
            # 测试集：good + 各类缺陷
            test_dir = base / "test"
            gt_dir = base / "ground_truth"
            for p in sorted((test_dir / "good").glob("*.png")):
                self.image_paths.append(str(p))
                self.mask_paths.append(None)
                self.labels.append(0)
            for defect in DEFECT_TYPES:
                for p in sorted((test_dir / defect).glob("*.png")):
                    mask_p = gt_dir / defect / (p.stem + "_mask.png")
                    self.image_paths.append(str(p))
                    self.mask_paths.append(str(mask_p) if mask_p.exists() else None)
                    self.labels.append(1)

    def __len__(self) -> int:
        return len(self.image_paths)

    def _load_image(self, path: str) -> np.ndarray:
        """OpenCV 读取 -> RGB -> resize 到 image_size×image_size。

        用 np.fromfile + imdecode 而非 cv2.imread：cv2.imread 在 Windows
        上无法读取含中文的绝对路径（ANSI 编码路径被截断）。
        """
        data = np.fromfile(path, dtype=np.uint8)
        img = cv2.imdecode(data, cv2.IMREAD_COLOR)
        if img is None:
            raise FileNotFoundError(f"图像读取失败：{path}")
        img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
        img = cv2.resize(img, (self.image_size, self.image_size), interpolation=cv2.INTER_LINEAR)
        return img

    def __getitem__(self, idx: int):
        image = self._load_image(self.image_paths[idx])
        label = self.labels[idx]
        mask = np.zeros((self.image_size, self.image_size), dtype=np.float32)
        if self.mask_paths[idx] is not None:
            # 同样用 imdecode 规避 cv2.imread 的中文路径问题
            m = cv2.imdecode(np.fromfile(self.mask_paths[idx], dtype=np.uint8), cv2.IMREAD_GRAYSCALE)
            m = cv2.resize(m, (self.image_size, self.image_size), interpolation=cv2.INTER_NEAREST)
            mask = (m > 0).astype(np.float32)

        return {
```

### 0.12 文件：`src/models/base.py`（模型统一接口）

```python
"""模型统一接口：所有模型实现 train / predict，输出统一的异常热力图与得分。"""

from __future__ import annotations

from abc import ABC, abstractmethod
from pathlib import Path

import numpy as np
import torch


class BaseAnomalyModel(ABC):
    """无监督异常检测模型基类。

    子类约定：
    - train(train_loader)：仅使用正常样本训练
    - predict_image(image)：输入单张归一化图像 (1,3,H,W) -> 原始异常热力图 (H,W) numpy
    - save / load：权重持久化
    """

    name: str = "base"

    def __init__(self, device: torch.device | str = "auto", image_size: int = 256):
        self.device = torch.device(
            device if device != "auto" else ("cuda" if torch.cuda.is_available() else "cpu")
        )
        self.image_size = image_size

    @abstractmethod
    def train(self, train_loader) -> None:
        """训练模型（只允许正常样本）。"""

    @abstractmethod
    def predict_image(self, image: torch.Tensor) -> np.ndarray:
        """单张图像推理。

        Args:
            image: (1,3,H,W) 归一化张量。

        Returns:
            原始异常热力图 (H,W) numpy，值越大越异常。
        """

    def predict(self, images: torch.Tensor) -> np.ndarray:
        """批量推理 -> (B,H,W) 原始异常热力图。"""
        maps = []
        for i in range(images.shape[0]):
            maps.append(self.predict_image(images[i : i + 1].to(self.device)))
        return np.stack(maps)

    def save(self, path: str | Path) -> None:
        path = Path(path)
        path.parent.mkdir(parents=True, exist_ok=True)
        state = self._state_dict()
        torch.save(state, path)
        print(f"[{self.name}] 权重已保存 -> {path}")

    def load(self, path: str | Path) -> None:
        path = Path(path)
        if not path.exists():
            raise FileNotFoundError(f"权重文件不存在：{path}")
        state = torch.load(path, map_location=self.device)
        self._load_state_dict(state)
        print(f"[{self.name}] 权重已加载 <- {path}")

    # ---- 子类可选覆写 ----
    def _state_dict(self) -> dict:
        raise NotImplementedError

    def _load_state_dict(self, state: dict) -> None:
        raise NotImplementedError
```

### 0.13 文件：`src/models/patchcore.py`（模型 1：PatchCore 主方案）

```python
"""PatchCore 模型（主方案，计划书模型 1）。

原理：冻结预训练 VGG16 仅作纹理特征提取器；
- 训练：提取正常样本局部 Patch 特征 -> 贪心 coreset 采样 -> FAISS 特征记忆库
- 推理：提取待测图 Patch 特征 -> k 近邻检索 -> 距离越大越偏离正常纹理 -> 像素级异常热力图

无需反向传播；支持单层 / 多层 VGG 特征融合（实验 2 消融）。
"""

from __future__ import annotations

import numpy as np
import torch
import torch.nn as nn
import torchvision

from .base import BaseAnomalyModel
from src.utils import resize_map, smooth_map

try:
    import faiss

    _FAISS_OK = True
except ImportError:  # pragma: no cover
    faiss = None
    _FAISS_OK = False

# VGG16 特征层名称 -> (features 序列下标, 通道数, 256 输入下的分辨率)
# 注意 torchvision vgg16.features 下标：14=conv3_3(256ch,32×32)、
# 24=conv4_3(512ch,16×16)、30=conv5_3(512ch,8×8)（下标 17 是 conv4_1，勿混淆）
VGG16_LAYERS = {
    "conv3_3": (14, 256, 32),
    "conv4_3": (24, 512, 16),
    "conv5_3": (30, 512, 8),
}


def _extract_layer_features(model: nn.Module, x: torch.Tensor, layer_name: str) -> torch.Tensor:
    """单次前向提取指定中间层特征图 (B,C,h,w)。"""
    idx, _, _ = VGG16_LAYERS[layer_name]
    out = {}
    hook = model.features[idx].register_forward_hook(
        lambda m, i, o: out.setdefault("feat", o)
    )
    try:
        with torch.no_grad():
            model(x)
    finally:
        hook.remove()
    return out["feat"]


def _extract_multilayer_features(model: nn.Module, x: torch.Tensor, layer_names: list[str]) -> list[torch.Tensor]:
    """一次前向同时提取多个中间层特征图 [(B,C,h,w), ...]（批量 GPU 推理用）。"""
    outs: dict[str, torch.Tensor] = {}
    handles = [
        model.features[VGG16_LAYERS[n][0]].register_forward_hook(
            lambda m, i, o, n=n: outs.setdefault(n, o)
        )
        for n in layer_names
    ]
    try:
        with torch.no_grad():
            model(x)
    finally:
        for h in handles:
            h.remove()
    return [outs[n] for n in layer_names]


class PatchCoreModel(BaseAnomalyModel):
    """PatchCore：VGG16 特征 + coreset 记忆库 + FAISS kNN 检索。"""

    name = "patchcore"

    def __init__(
        self,
        device="auto",
        image_size: int = 256,
        layers: list[str] | str | None = None,
        k: int = 5,
        coreset_ratio: float = 0.1,
        stride: int = 1,
    ):
        super().__init__(device, image_size)
        if not _FAISS_OK:
            raise ImportError("PatchCore 需要 faiss-cpu，请先 pip install faiss-cpu")
        self.layers = (
            [layers] if isinstance(layers, str) else (layers or ["conv4_3"])
        )
        for name in self.layers:
            if name not in VGG16_LAYERS:
                raise ValueError(f"未知 VGG 特征层 {name}，可选 {list(VGG16_LAYERS)}")
        self.k, self.coreset_ratio, self.stride = k, coreset_ratio, stride

        # 冻结预训练 VGG16（仅特征提取，无梯度）
        self.backbone = torchvision.models.vgg16(
            weights=torchvision.models.VGG16_Weights.IMAGENET1K_V1
        ).to(self.device)
        self.backbone.eval()
        for p in self.backbone.parameters():
            p.requires_grad = False

        self.memory_bank: np.ndarray | None = None  # (M,C) 记忆库
        self.index: faiss.IndexFlatL2 | None = None
        self._bank_t: torch.Tensor | None = None   # GPU 记忆库缓存（批量 GPU 检索用）

    # ------------------------------------------------------------------ #
    # 特征提取
    # ------------------------------------------------------------------ #
    @torch.no_grad()
    def _extract_patches(self, images: torch.Tensor) -> np.ndarray:
        """提取多层融合 Patch 特征。

        Args:
            images: (B,3,H,W) 归一化图像。

        Returns:
            (B*P, C) 特征矩阵，P 为特征图 Patch 数（多层取最小分辨率并拼接）。
        """
        batch_feats = []
        for img in images:
            img = img.unsqueeze(0).to(self.device)
            layer_feats = []
            resolutions = [VGG16_LAYERS[n][2] for n in self.layers]
            target_res = min(resolutions)
            for name in self.layers:
                feat = _extract_layer_features(self.backbone, img, name)  # (1,C,h,w)
                if feat.shape[-1] != target_res:
                    feat = torch.nn.functional.avg_pool2d(
                        feat, feat.shape[-1] // target_res
                    )
                layer_feats.append(feat)
            fused = torch.cat(layer_feats, dim=1)  # (1,Cs,h,w) 多层特征融合
            # 步长采样（计划书：patch 采样步长超参数）
            fused = fused[:, :, :: self.stride, :: self.stride]
            batch_feats.append(fused.reshape(fused.shape[1], -1).T)  # (P,C)
        return torch.cat(batch_feats, dim=0).cpu().numpy()

    # ------------------------------------------------------------------ #
    # 训练：构建 coreset 记忆库
    # ------------------------------------------------------------------ #
    def train(self, train_loader) -> None:
        print(f"[{self.name}] 提取训练特征（layers={self.layers}, stride={self.stride}）...")
        all_feats = []
        for batch in train_loader:
            all_feats.append(self._extract_patches(batch["image"]))
        feats = np.concatenate(all_feats, axis=0)
        print(f"[{self.name}] 特征总数 {len(feats)} × {feats.shape[1]} 维")

        n_coreset = max(1, int(len(feats) * self.coreset_ratio))
        self.memory_bank = _greedy_coreset(feats, n_coreset)
        print(f"[{self.name}] coreset 采样 {len(self.memory_bank)} 个 Patch（比例 {self.coreset_ratio}）")

        # 构建 FAISS 最近邻索引
        self.index = faiss.IndexFlatL2(self.memory_bank.shape[1])
        self.index.add(self.memory_bank.astype(np.float32))
        self._bank_t = None  # 记忆库已变，失效 GPU 缓存
        print(f"[{self.name}] FAISS 记忆库构建完成")

    # ------------------------------------------------------------------ #
    # 推理
    # ------------------------------------------------------------------ #
    def _extract_patches_batch(self, images: torch.Tensor) -> torch.Tensor:
        """批量提取多层融合 Patch 特征（一次前向，全部图像并行）。

        Args:
            images: (B,3,H,W) 归一化图像。

        Returns:
            (B,P,C) 特征，P 为特征图 Patch 数（多层取最小分辨率并拼接）。
        """
        images = images.to(self.device)
        layer_feats = _extract_multilayer_features(self.backbone, images, self.layers)
        resolutions = [VGG16_LAYERS[n][2] for n in self.layers]
        target_res = min(resolutions)
        for i, feat in enumerate(layer_feats):
            if feat.shape[-1] != target_res:
                feat = torch.nn.functional.avg_pool2d(feat, feat.shape[-1] // target_res)
                layer_feats[i] = feat
        fused = torch.cat(layer_feats, dim=1)  # (B,Cs,h,w)
        fused = fused[:, :, :: self.stride, :: self.stride]
        b, c, h, w = fused.shape
        return fused.permute(0, 2, 3, 1).reshape(b, h * w, c)  # (B,P,C)

    def _get_bank_tensor(self) -> torch.Tensor:
        """GPU 记忆库（惰性缓存）。

        注意：比较 device.type 而非整个 device——self.device 为
        ``torch.device("cuda")``（index=None），而实际张量在 ``cuda:0``，
        直接比较不相等会导致每次调用都重建记忆库（173MB 拷贝 + 传输）。
        """
        if self._bank_t is None or self._bank_t.device.type != self.device.type:
            self._bank_t = torch.from_numpy(self.memory_bank.astype(np.float32)).to(self.device)
        return self._bank_t

    def _knn_gpu(self, feats_t: torch.Tensor, k: int) -> torch.Tensor:
        """GPU 分块 L2 距离 + k 近邻（数学与 FAISS IndexFlatL2 一致）。

        Args:
            feats_t: (Q,C) 查询特征（GPU）。

        Returns:
            (Q,k) k 近邻距离（未开方，与 FAISS 返回的 squared L2 一致）。
        """
        bank = self._get_bank_tensor()  # (M,C)
        m = bank.shape[0]
        feats_norm = feats_t.pow(2).sum(1, keepdim=True)  # (Q,1)
        bank_norm = bank.pow(2).sum(1)  # (M,)，只读一遍记忆库
        # 查询数小时整库一次 GEMM（少分块 = 少 kernel 启动与归约，实测快 3 倍以上）；
        # 查询数大时按记忆库分块，控制 d² 中间矩阵显存峰值（约 256MB）
        max_elems = 1 << 26
        bank_chunk = m if feats_t.shape[0] * m <= max_elems else max_elems // feats_t.shape[0]
        cand_d = []
        for i in range(0, m, bank_chunk):
            b = bank[i : i + bank_chunk]  # (chunk,C)
            # d² = |x|² + |y|² − 2·x·yᵀ：矩阵乘法走 Tensor Core，
            # 比 cdist（逐输出元素重读输入）快一个数量级
            d2 = feats_norm + bank_norm[i : i + bank_chunk].unsqueeze(0) - 2.0 * (feats_t @ b.t())
            d2 = d2.clamp_min(0.0)  # 防展开式浮点误差出现微小负数
            if k == 1:
                cand_d.append(d2.min(dim=1).values)  # k=1 行最小即最近邻距离
            else:
                top_d, _ = torch.topk(-d2, min(k, d2.shape[1]), dim=1)
                cand_d.append(-top_d)
        if k == 1:
            return torch.stack(cand_d, dim=1).min(dim=1).values.unsqueeze(1)
        cand_d = torch.cat(cand_d, dim=1)  # (Q, 块数*k)
        final_d, _ = torch.topk(-cand_d, k, dim=1)  # 各块候选合并后取最终 k
        return -final_d

    def _amaps_from_feats(self, feats_t: torch.Tensor, n: int) -> np.ndarray:
        """(B,P,C) 特征 -> kNN 距离 -> (B,H,W) 异常热力图（GPU 检索）。"""
        q, _, c = feats_t.shape
        flat = feats_t.reshape(-1, c)
        scores = []
        for i in range(0, flat.shape[0], 2048):  # 查询分块，控制显存峰值
            scores.append(self._knn_gpu(flat[i : i + 2048], self.k).mean(dim=1))
        scores = torch.cat(scores)  # (B*P,)
        target_res = min(VGG16_LAYERS[n][2] for n in self.layers) // self.stride
        maps = scores.reshape(n, target_res, target_res).cpu().numpy()
        out = np.stack([
            smooth_map(resize_map(m, (self.image_size, self.image_size)), sigma=4)
            for m in maps
        ])
        return out

    def predict_image_faiss(self, image: torch.Tensor) -> np.ndarray:
        """FAISS CPU 检索单图推理（官方评估协议路径，跨设备确定性一致）。"""
        assert self.index is not None, "请先调用 train() 构建记忆库"
        feats = self._extract_patches(image)  # (P,C)
        distances, _ = self.index.search(feats.astype(np.float32), self.k)
        anomaly_scores = distances.mean(axis=1)  # (P,)

        # 还原成特征图空间分辨率
        target_res = min(VGG16_LAYERS[n][2] for n in self.layers) // self.stride
        amap = anomaly_scores.reshape(target_res, target_res)
        amap = resize_map(amap, (self.image_size, self.image_size))
        return smooth_map(amap, sigma=4)

    def predict_image(self, image: torch.Tensor) -> np.ndarray:
        """kNN 平均距离 -> 异常热力图（GPU 分块 GEMM 检索，与 FAISS 数学等价）。"""
        assert self.index is not None, "请先调用 train() 构建记忆库"
        if self.device.type == "cuda":
            feats_t = self._extract_patches_batch(image)  # (1,P,C) 一次前向
            return self._amaps_from_feats(feats_t, 1)[0]
        return self.predict_image_faiss(image)  # CPU 回退

    def predict(self, images: torch.Tensor) -> np.ndarray:
        """批量推理 -> (B,H,W) 异常热力图（批量特征提取 + GPU 检索）。"""
        assert self.index is not None, "请先调用 train() 构建记忆库"
        if self.device.type == "cuda":
            feats_t = self._extract_patches_batch(images)
            return self._amaps_from_feats(feats_t, images.shape[0])
        return super().predict(images)

    # ------------------------------------------------------------------ #
    # 持久化
    # ------------------------------------------------------------------ #
    def _state_dict(self) -> dict:
        return {
            "memory_bank": self.memory_bank,
            "layers": self.layers,
            "k": self.k,
            "coreset_ratio": self.coreset_ratio,
            "stride": self.stride,
        }

    def _load_state_dict(self, state: dict) -> None:
        self.memory_bank = state["memory_bank"]
        self.layers = state["layers"]
        self.k = state["k"]
        self.coreset_ratio = state["coreset_ratio"]
        self.stride = state["stride"]
        self.index = faiss.IndexFlatL2(self.memory_bank.shape[1])
        self.index.add(self.memory_bank.astype(np.float32))
        self._bank_t = None  # 记忆库已变，失效 GPU 缓存


def _greedy_coreset(features: np.ndarray, n: int) -> np.ndarray:
    """贪心最远点采样（k-center-greedy 近似）压缩记忆库。"""
    feats = features.astype(np.float32)
    if n >= len(feats):
        return feats
    rng = np.random.default_rng(42)
    idx = faiss.IndexFlatL2(feats.shape[1])
    first = int(rng.integers(len(feats)))
    idx.add(feats[first : first + 1])
    selected = [first]
    remaining = n - 1
    # 每次迭代取离已选集最远的点加入（分批加速）
    batch = max(8, remaining // 100)
    while len(selected) < n:
        d, _ = idx.search(feats, 1)
        order = np.argsort(-d[:, 0])
        add = []
        for i in order:
            if len(selected) + len(add) >= n:
                break
            add.append(int(i))
        idx.add(feats[add])
        selected.extend(add)
        if len(add) == 0:
            break
    return feats[selected[:n]]
```

```python
原理：无需真实缺陷样本；
- 训练：对正常图像随机剪切粘贴生成伪异常（CutPaste / CutPaste-Scar），
  与正常样本做二分类自监督训练（ResNet18）
- 推理：补丁级滑窗分类（原论文定位方案）或异常类 GradCAM 生成热力图

import torch
import torch.nn as nn
import torch.nn.functional as F
import torchvision

# ---------------------------------------------------------------------- #
# CutPaste 伪异常增广（GPU 向量化 tensor 实现，直接在归一化张量上操作）
# ---------------------------------------------------------------------- #
def cutpaste_single(x: torch.Tensor, variant: str = "normal") -> torch.Tensor:
    """对单张归一化张量 (3,H,W) 执行 CutPaste 剪切粘贴，返回新图像。

    注意：使用模块级 np.random（受 set_seed 控制），保证实验可复现。
    """
    h, w = x.shape[1], x.shape[2]
    area_ratio = float(np.random.uniform(0.02, 0.15))

    if variant == "scar":
        # 细长划痕状补丁（CutPaste-Scar）
        patch_h = int(np.random.uniform(2, 4))
        patch_w = max(2, int(area_ratio * h * w / patch_h))
    else:
        # 常规矩形补丁
        ratio = float(np.random.uniform(0.3, 3.3))
        patch_h = int(np.sqrt(area_ratio * ratio) * h)
        patch_w = int(np.sqrt(area_ratio / ratio) * w)
    patch_h = min(max(patch_h, 2), h - 1)
    patch_w = min(max(patch_w, 2), w - 1)

    x1 = int(np.random.randint(0, w - patch_w + 1))
    y1 = int(np.random.randint(0, h - patch_h + 1))
    x2 = int(np.random.randint(0, w - patch_w + 1))
    y2 = int(np.random.randint(0, h - patch_h + 1))

    patch = x[:, y1 : y1 + patch_h, x1 : x1 + patch_w].clone()
    # 模糊（GPU avg_pool 近似高斯）：避免分类器走「硬边缘」捷径（原论文策略）；
    # 模糊核按补丁尺寸自适应（划痕补丁可能只有 2~3 像素高）
    k = patch_h if patch_h % 2 == 1 else patch_h - 1
    k = min(k, 15)
    if k > 1:
        patch = F.avg_pool2d(patch.unsqueeze(0), kernel_size=k, stride=1, padding=k // 2)[0]
    # 补丁亮度扰动（归一化空间小幅度加性扰动）
    patch = patch + float(np.random.uniform(-0.05, 0.05))

    out = x.clone()
    out[:, y2 : y2 + patch_h, x2 : x2 + patch_w] = patch
    # 全图亮度/对比度抖动（原论文：迫使分类器学习纹理一致性而非局部伪影）
    out = out * float(np.random.uniform(0.92, 1.08))
    out = out + float(np.random.uniform(-0.03, 0.03))
    return out


def cutpaste_batch(x: torch.Tensor) -> torch.Tensor:
    """批量生成伪异常（一半 CutPaste，一半 CutPaste-Scar）。"""
    out = []
    for i in range(x.shape[0]):
        variant = "normal" if i % 2 == 0 else "scar"
        out.append(cutpaste_single(x[i], variant))
    return torch.stack(out)


# ---------------------------------------------------------------------- #
# GradCAM
# ---------------------------------------------------------------------- #
class _GradCAM:
    """ResNet 最后一层卷积的 GradCAM。"""

    def __init__(self, model: nn.Module, target: nn.Module):
        self.activations, self.gradients = None, None
        self.fwd = target.register_forward_hook(self._fwd)
        # 用 full_backward_hook：BasicBlock 前向含多个 autograd 节点，
        # 旧版非完整钩子可能拿不到梯度
        self.bwd = target.register_full_backward_hook(self._bwd)

    def _fwd(self, module, i, o):
        self.activations = o.detach()

    def _bwd(self, module, gi, go):
        self.gradients = go[0].detach()

    def __call__(self, score: torch.Tensor):
        self.gradients = None
        score.backward()
        w = self.gradients.mean(dim=(2, 3), keepdim=True)  # 梯度权重
        cam = F.relu((w * self.activations).sum(dim=1, keepdim=True))  # (1,1,h,w)
        return cam

    def remove(self):
        self.fwd.remove()
        self.bwd.remove()

# CutPaste 模型

class CutPasteModel(BaseAnomalyModel):
    """CutPaste：伪异常自监督二分类 + GradCAM 定位。"""

    name = "cutpaste"

    def __init__(
        self,
        device="auto",
        image_size: int = 256,
        epochs: int = 50,
        lr: float = 0.01,
        patience: int = 8,
        min_epochs: int = 10,  # 早停保护：epoch 数不足不保存模型（防止欠拟合最优被选中）
        loc_mode: str = "gradcam",  # 定位方式：gradcam=异常类GradCAM（默认）/ patch=补丁级分类
        patch_size: int = 32,
    ):
        super().__init__(device, image_size)
        self.epochs, self.lr, self.patience, self.min_epochs = epochs, lr, patience, min_epochs
        self.loc_mode = loc_mode
        self.patch_size = patch_size
        self.net = torchvision.models.resnet18(
            weights=torchvision.models.ResNet18_Weights.IMAGENET1K_V1
        )
        self.net.fc = nn.Linear(self.net.fc.in_features, 2)  # 正常 / 伪异常
        self.net = self.net.to(self.device)
        self._cam: _GradCAM | None = None

    # ------------------------------------------------------------------ #
    # 训练
    # ------------------------------------------------------------------ #
    def train(self, train_loader) -> None:
        net = self.net.train()
        optimizer = torch.optim.SGD(net.parameters(), lr=self.lr, momentum=0.9, weight_decay=1e-4)
        criterion = nn.CrossEntropyLoss()

        # 切 10% 验证集做早停
        images = torch.cat([b["image"] for b in train_loader], dim=0)
        n_val = max(1, len(images) // 10)
        val_images, train_images = images[:n_val], images[n_val:]

        best_loss, best_state, bad_epochs = float("inf"), None, 0
        n = len(train_images)
        print(f"[{self.name}] 二分类训练：正常 {n} + 伪异常 {n}，epochs={self.epochs}")
        for epoch in range(1, self.epochs + 1):
            perm = torch.randperm(n)
            total = 0.0
            for i in range(0, n, 32):
                xb = train_images[perm[i : i + 32]].to(self.device)
                pseudo = self._make_pseudo(xb)
                inputs = torch.cat([xb, pseudo], dim=0)
                labels = torch.cat(
                    [torch.zeros(len(xb)), torch.ones(len(pseudo))]
                ).long().to(self.device)
                logits = net(inputs)
                loss = criterion(logits, labels)
                optimizer.zero_grad()
                loss.backward()
                optimizer.step()
                total += loss.item() * len(inputs)
            train_loss = total / (2 * n)

            with torch.no_grad():
                vp = self._make_pseudo(val_images.to(self.device))
                v_inputs = torch.cat([val_images.to(self.device), vp], dim=0)
                v_labels = torch.cat(
                    [torch.zeros(len(val_images)), torch.ones(len(vp))]
                ).long().to(self.device)
                val_loss = criterion(net(v_inputs), v_labels).item()

            if val_loss < best_loss - 1e-5:
                best_loss, bad_epochs = val_loss, 0
                # min_epochs 之前不保存，避免选中欠拟合的最优
                if epoch >= self.min_epochs:
                    best_state = {k: v.clone() for k, v in net.state_dict().items()}
            else:
                bad_epochs += 1
            if epoch % 10 == 0:
                print(f"  epoch {epoch:3d} | train loss {train_loss:.5f} | val loss {val_loss:.5f}")
            if best_state is not None and bad_epochs >= self.patience:
                print(f"[{self.name}] 早停于 epoch {epoch}")
                break

        if best_state is None:
            best_state = {k: v.clone() for k, v in net.state_dict().items()}
        net.load_state_dict(best_state)
        net.eval()
        print(f"[{self.name}] 训练完成，最佳 val loss = {best_loss:.5f}")

    def _make_pseudo(self, xb: torch.Tensor) -> torch.Tensor:
        """直接在归一化张量上生成伪异常（GPU 向量化，无 uint8 往返）。"""
        return cutpaste_batch(xb)

    # ------------------------------------------------------------------ #
    # 推理：异常热力图
    # ------------------------------------------------------------------ #
    def predict_image(self, image: torch.Tensor) -> np.ndarray:
        self.net.eval()
        x = image.to(self.device)
        if self.loc_mode == "gradcam":
            return self._predict_gradcam(x)
        return self._predict_patch(x)

    def _predict_patch(self, x: torch.Tensor) -> np.ndarray:
        """补丁级定位（原论文做法）：滑窗切块逐块二分类。

        异常得分 = 1 - P(正常)。分类器在整图正常纹理上训练，正常区域补丁
        得分高、缺陷区域补丁得分低，拼成网格图即为异常热力图。
        """
        ps = self.patch_size
        stride = max(ps // 2, 4)
        unfold = F.unfold(x, kernel_size=ps, stride=stride)  # (1, 3*ps*ps, L)
        L = unfold.shape[-1]
        patches = unfold[0].t().reshape(L, 3, ps, ps)
        p_normal = []
        with torch.no_grad():
            for i in range(0, L, 64):
                logits = self.net(patches[i : i + 64])
                p_normal.append(F.softmax(logits, dim=1)[:, 0])
        p_normal = torch.cat(p_normal).cpu().numpy()  # (L,)
        gh = (self.image_size - ps) // stride + 1  # 网格边长（输入为方形）
        amap = (1.0 - p_normal).reshape(gh, gh)
        amap = resize_map(amap, (self.image_size, self.image_size))
        return smooth_map(amap, sigma=4)

    def _predict_gradcam(self, x: torch.Tensor) -> np.ndarray:
        """GradCAM 定位 + 分类概率校准（原论文 CutPaste+GradCAM 方案）。

        - 定位：对异常类得分反向传播，突出疑似伪异常（不连续纹理）区域
        - 校准：cam 逐图除以自身最大值再乘以 P(异常)，使热力图峰值等于
          分类器异常概率——图像级得分（取最大值）与论文的分类概率一致，
          同时保留 GradCAM 的空间分布用于像素级定位
        """
        if self._cam is None:
            # 必须在 forward 之前注册钩子，否则捕获不到激活与梯度
            self._cam = _GradCAM(self.net, self.net.layer4[-1])
        x = x.detach().requires_grad_(True)
        # GradCAM 依赖反向传播，须显式开启梯度（app.detect 外层统一
        # torch.no_grad()，此处 enable_grad 优先于外层生效）
        with torch.enable_grad():
            logits = self.net(x)
            p_anomaly = F.softmax(logits, dim=1)[0, 1].item()
            cam = self._cam(logits[:, 1].sum())  # 异常类得分反向传播
            cam = cam.detach()
        cam = F.interpolate(cam, size=(self.image_size, self.image_size), mode="bilinear")[0, 0]
        cam = smooth_map(cam.detach().cpu().numpy(), sigma=4)
        cam_max = cam.max()
        if cam_max < 1e-8:
            # 无定位信号（如自信判为正常的图）：均匀图，峰值仍为 P(异常)，
            # 保证图像级得分与分类概率完全一致
            return np.full_like(cam, p_anomaly)
        # 先平滑后缩放：峰值精确等于 P(异常)，图像级得分与分类概率一致
        return cam / cam_max * p_anomaly

    # ------------------------------------------------------------------ #
    # 持久化
    # ------------------------------------------------------------------ #
    def _state_dict(self) -> dict:
        return {"net": self.net.state_dict(), "config": {"epochs": self.epochs, "lr": self.lr}}

    def _load_state_dict(self, state: dict) -> None:
        self.net.load_state_dict(state["net"])
```

```python
def _patch_gaussian_fit() -> None:
    """修复 MultiVariateGaussian 在 Windows + CUDA 上的性能问题。

    anomalib 2.6.2 原始实现：逐位置 Python 循环求协方差，随后
    ``torch.linalg.inv`` 对 (P,550,550) 批量求逆。cuSOLVER 批量求逆在本机
    （RTX 4060 Laptop + Windows）逐批串行执行，实测 20+ 分钟未完成
    （GPU 利用率 100% 但功耗仅 ~37W，纯内核启动开销）。
    本补丁：einsum 向量化协方差 + 转 CPU numpy 求逆（约 15~30 秒）。
    数学等价：ddof=1 与 0.01+1e-5 对角线正则均与原实现一致。
    """
    from anomalib.models.components import MultiVariateGaussian

    def _fast_forward(self, embedding: torch.Tensor) -> list[torch.Tensor]:
        device = embedding.device
        b, c, h, w = embedding.size()
        vectors = embedding.view(b, c, h * w)
        mean = torch.mean(vectors, dim=0)  # (C, P)
        centered = vectors - mean.unsqueeze(0)  # (B, C, P)
        # 向量化协方差 (P, C, C)，ddof=1（与 anomalib _cov 默认一致）
        cov = torch.einsum("bcp,bdp->pcd", centered, centered) / (b - 1)
        identity = torch.eye(c, device=device)
        cov = cov + (0.01 + 1e-5) * identity  # 与原实现相同的对角线正则
        cov_np = cov.detach().cpu().numpy()
        del cov  # 立即释放 GPU 显存
        inv = np.stack([np.linalg.inv(cov_np[i]) for i in range(cov_np.shape[0])])
        self.mean = mean
        self.inv_covariance = torch.from_numpy(inv).to(device)
        return [self.mean, self.inv_covariance]

    MultiVariateGaussian.forward = _fast_forward

        category = train_loader.dataset.category
        self._category = category

        # Anomalib 内部校验拒绝非 ASCII 路径，改用 junction 链接
        self._anomalib_root = _ensure_ascii_link(self._data_root)

        datamodule = MVTecAD(
            root=str(self._anomalib_root),
            category=category,
            train_batch_size=32,
            eval_batch_size=32,
            num_workers=4,
            seed=42,
        )
        model = Padim(
            backbone=self.backbone,
            layers=self.layers,
            pre_trained=True,
            n_features=self.n_features,
        )
        accelerator = "gpu" if torch.cuda.is_available() else "cpu"
        engine = Engine(accelerator=accelerator, devices=1, max_epochs=1)

        print(f"[{self.name}] Anomalib PaDiM 训练（backbone={self.backbone}, layers={self.layers}）")
        engine.fit(model, datamodule=datamodule)
        self._model, self._engine, self._datamodule = model, engine, datamodule
        print(f"[{self.name}] 高斯统计量拟合完成")

    # ------------------------------------------------------------------ #
    # 推理：整测试集预测 -> {path: anomaly_map}
    # ------------------------------------------------------------------ #
    def predict_dataset(self) -> dict[str, np.ndarray]:
        """对整个测试集推理，返回 {图像绝对路径: 原始异常热力图 (H,W)}。"""
        if self._pred_cache is not None:
            return self._pred_cache
        if self._model is None:
            if self._loaded_stats is None:
                raise RuntimeError("请先调用 train()")
            self._rebuild_from_checkpoint()
        preds = self._engine.predict(
            self._model, datamodule=self._datamodule, return_predictions=True
        )

        result: dict[str, np.ndarray] = {}
        batches = preds if isinstance(preds, (list, tuple)) else [preds]
        for batch in batches:
            for i in range(len(batch)):
                path = str(batch.image_path[i])
                # junction 路径映射回真实数据根，保证与自定义数据集路径对齐
                link = str(self._anomalib_root)
                if path.startswith(link):
                    path = str(self._data_root) + path[len(link):]
                amap = batch.anomaly_map[i]
                if isinstance(amap, torch.Tensor):
                    amap = amap.detach().cpu().numpy()
                result[path] = np.asarray(amap, dtype=np.float32)
        self._pred_cache = result
        return result

    def predict_dataset_fast(self, batch_size: int = 32) -> dict[str, np.ndarray]:
        """批量前向 + GPU 马氏距离推理（优化实验：绕开 Engine 调度与 dataloader 开销）。

        与 predict_dataset() 输出同源：同一 PadimModel、同一高斯统计量、
        同一预处理（read_image float32 → v2 抗锯齿 resize 256 → ImageNet
        归一化），仅省去 Engine/Lightning 的调度开销，单张耗时降至毫秒级。
        """
        if self._pred_cache is not None:
            return self._pred_cache
        if self._model is None:
            if self._loaded_stats is None:
                raise RuntimeError("请先调用 train()")
            self._rebuild_from_checkpoint()
        from anomalib.data.utils.image import read_image
        from torchvision.transforms import v2 as T

        torch_model = self._model.model.to(self.device)
        was_training = torch_model.training
        torch_model.eval()
        pp = T.Compose([
            T.Resize((256, 256), antialias=True),
            T.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
        ])
        # _data_root 是类别目录的父级（MVTec 根），需拼接 category
        test_root = Path(self._data_root) / self._category / "test"
        paths = sorted(
            p for p in test_root.glob("*/*")
            if p.suffix.lower() in (".png", ".jpg", ".bmp")
        )
        result: dict[str, np.ndarray] = {}
        try:
            for i in range(0, len(paths), batch_size):
                chunk = paths[i : i + batch_size]
                imgs = [read_image(str(p), as_tensor=True) for p in chunk]  # float32 [0,1]
                x = torch.stack([pp(t) for t in imgs]).to(self.device)
                with torch.no_grad():
                    out = torch_model(x)
                for p, amap in zip(chunk, out.anomaly_map):
                    result[str(p)] = np.asarray(amap.detach().cpu(), dtype=np.float32)
        finally:
            if was_training:
                torch_model.train()
        self._pred_cache = result
        return result

    def predict_image(self, image: torch.Tensor) -> np.ndarray:
        """单张图像推理（Gradio 三模型部署）：返回原始异常热力图 (H, W)。

        复用 Anomalib PadimModel 的 eval 前向：特征提取 → 随机投影 →
        马氏距离 → 上采样求和。输入须按 Anomalib 默认 PreProcessor 预处理
        （torchvision v2 抗锯齿 resize 256×256 + ImageNet 归一化），
        Gradio 端 app.py 的 ``_preprocess_tv`` 即此实现，与整测试集评估同源。
        """
        if self._model is None:
            if self._loaded_stats is None:
                raise RuntimeError("请先调用 train()")
            self._rebuild_from_checkpoint()
        # 直接调用前向不走 Engine.predict，需手动把子模块搬到设备上
        # （重建时权重与统计量仍在 CPU，Engine 路径由 Lightning 自动搬移）；
        # 已在设备上时跳过 .to()（遍历全部参数张量有毫秒级开销，热路径每张图都会调用）
        torch_model = self._model.model
        if next(torch_model.parameters()).device.type != self.device.type:
            torch_model = torch_model.to(self.device)
        was_training = torch_model.training
        torch_model.eval()
        with torch.no_grad():
            out = torch_model(image.to(self.device))
        if was_training:
            torch_model.train()
        amap = out.anomaly_map
        if isinstance(amap, torch.Tensor):
            amap = amap.detach().cpu().numpy()
        return np.asarray(amap, dtype=np.float32).squeeze()

    # ------------------------------------------------------------------ #
    # 持久化：保存已拟合的高斯统计量（mean + 逆协方差），
    # 加载后无需重新拟合即可直接推理（PaDiM 的"权重"即统计量）
    # ------------------------------------------------------------------ #
    def _state_dict(self) -> dict:
        state = {"backbone": self.backbone, "layers": self.layers, "n_features": self.n_features}
        if self._model is not None:
            gaussian = self._model.model.gaussian
            state["category"] = self._category
            state["data_root"] = str(self._data_root)
            state["mean"] = gaussian.mean.detach().cpu()
            state["inv_covariance"] = gaussian.inv_covariance.detach().cpu()
            # idx 随机投影与统计量必须成对保存，否则加载后投影不一致导致指标失真
            state["idx"] = self._model.model.idx.detach().cpu()
        return state

    def _load_state_dict(self, state: dict) -> None:
        self.backbone = state["backbone"]
        self.layers = state["layers"]
        self.n_features = state.get("n_features")
        self._loaded_stats = None
        if "mean" in state:
            self._loaded_stats = (
                state["data_root"],
                state["mean"],
                state["inv_covariance"],
                state["idx"],
            )
            self._category = state.get("category", "metal_nut")

    def _rebuild_from_checkpoint(self) -> None:
        """从 checkpoint 恢复模型与高斯统计量（等价于 train() 后的状态）。"""
        from anomalib.data import MVTecAD
        from anomalib.engine import Engine
        from anomalib.models import Padim

        data_root, mean, inv_cov, idx = self._loaded_stats
        self._data_root = Path(data_root)
        self._anomalib_root = _ensure_ascii_link(self._data_root)

        model = Padim(
            backbone=self.backbone,
            layers=self.layers,
            pre_trained=True,
            n_features=self.n_features,
        )
        gaussian = model.model.gaussian
        device = model.device
        gaussian.mean = mean.to(device)
        gaussian.inv_covariance = inv_cov.to(device)
        # 恢复训练时的随机投影索引（与统计量成对，缺一不可）
        model.model.idx.copy_(idx.to(model.model.idx.device))

        self._datamodule = MVTecAD(
            root=str(self._anomalib_root),
            category=self._category,
            train_batch_size=32,
            eval_batch_size=32,
            num_workers=4,
            seed=42,
        )
        accelerator = "gpu" if torch.cuda.is_available() else "cpu"
        self._engine = Engine(accelerator=accelerator, devices=1, max_epochs=1)
        self._model = model
        print(f"[{self.name}] 已从 checkpoint 恢复高斯统计量（跳过拟合）")
```

```python
"""实验 5：PatchCore 主方案优化搜索（全量记忆库下 k 值与特征层组合）。

优化配置（coreset 0.1 + k=5，0.6994/0.5395）——220 张小训练集上
coreset 压缩损失代表性、k=5 平均稀释最近邻信号。

本实验在全量记忆库（coreset_ratio=1.0）、无数据增强的已验证最优状态下，
继续搜索 k ∈ {1,3,5} 与特征层（单层 conv4_3 / 多层融合）组合，
确定主方案的最终配置。

用法：python experiments/exp5_patchcore_optimize.py [--retrain]

# (标签, 特征层, k, 复用已有权重文件名或 None)
SWEEP_GROUPS = [
    ("conv4_3_k1", ["conv4_3"], 1, "patchcore_baseline.pth"),  # 复用实验 4 基线权重
    ("conv4_3_k3", ["conv4_3"], 3, None),

    ("多层融合_k5", ["conv3_3", "conv4_3"], 5, None),

def evaluate_auroc_only(model: PatchCoreModel, test_loader) -> tuple[float, float]:
    """仅计算图像级 / 像素级 AUROC（不做 20 次重复耗时测量，加速搜索）。

    协议与 src/evaluation.py 完全一致：图像级得分 = 原始热力图最大值；
    像素级 AUROC = 逐图 min-max 归一化后全局展平。
    """
    raw_maps, masks, labels = [], [], []
    for batch in test_loader:
        maps = model.predict(batch["image"].to(model.device))  # (B,H,W)
        raw_maps.append(maps)
        masks.append(batch["mask"].squeeze(1).numpy())
        labels.append(batch["label"].numpy())
    raw_maps = np.concatenate(raw_maps, axis=0)
    masks = np.concatenate(masks, axis=0)
    labels = np.concatenate(labels, axis=0)
    scores = raw_maps.max(axis=(1, 2))
    return (compute_image_auroc(scores, labels),
            compute_pixel_auroc(raw_maps, masks))

            device="auto",
            image_size=cfg["dataset"]["image_size"],
            layers=layers,
            k=k,
            coreset_ratio=1.0,        # 全量记忆库：无 coreset 压缩

        ckpt = reuse_ckpt or f"patchcore_opt_{tag}.pth"
```

## 第 6 部分：Gradio 演示系统（三模型推理 + 质检判定）

### 6.1 文件：`app.py`

```python
"""基于纹理特征的工业缺陷检测系统 — Gradio 交互入口（计划书 4.4）。

模块划分：
1. 图像上传与预处理：本地图片上传 -> RGB -> resize 256×256 -> ImageNet 归一化
2. 三模型推理：PatchCore / PaDiM / CutPaste 可切换
3. 质检判定（参考 B2 组螺丝缺陷检测系统）：异常得分 ≥ 判定阈值判「不合格」，
   阈值 = 灵敏度倍数 × 自动阈值（训练集正常样本统计，随权重分发），
   默认 1.0 即自动阈值；倍数调小更严格（少漏检）、调大更宽松（少误报）
4. 可视化输出：原图、异常热力图（叠加）、缺陷掩码（≥ 判定阈值区域）、
   合格/不合格彩色判定与异常得分

启动：python app.py   然后浏览器打开 http://127.0.0.1:7860
"""

from __future__ import annotations

import json
import sys
from pathlib import Path

# 保证在任意目录启动时都能导入 src
sys.path.insert(0, str(Path(__file__).resolve().parent))

import cv2
import gradio as gr
import numpy as np
import torch
from PIL import Image
from torchvision import transforms
from torchvision.transforms import v2 as T

from src import CHECKPOINT_DIR
from src.models import CutPasteModel, PadimAnomalibModel, PatchCoreModel
from src.utils import IMAGENET_MEAN, IMAGENET_STD, heatmap_overlay, set_seed

DEVICE = "cuda" if torch.cuda.is_available() else "cpu"

# ---------------------------------------------------------------------- #
# 模型加载
# ---------------------------------------------------------------------- #
# 显示名 -> (模型类, 权重文件名)
MODEL_CLASSES = {
    "PatchCore（主方案）": (PatchCoreModel, "patchcore.pth"),
    "PaDiM": (PadimAnomalibModel, "padim.pth"),
    "CutPaste": (CutPasteModel, "cutpaste.pth"),
}

_MODELS: dict[str, object] = {}
_AUTO_THRESHOLDS: dict[str, float] = {}


def load_models() -> dict[str, str]:
    """启动时加载全部可用模型权重与自动判定阈值，返回提示信息。"""
    ckpt_dir = Path(CHECKPOINT_DIR)
    messages = []
    for name, (cls, ckpt_name) in MODEL_CLASSES.items():
        ckpt = ckpt_dir / ckpt_name
        try:
            model = cls(device=DEVICE)
            model.load(ckpt)
            _MODELS[name] = model
        except FileNotFoundError:
            messages.append(f"⚠ 未找到 {name} 权重（{ckpt.name}），"
                            f"请先运行：python experiments/exp1_model_compare.py")
    tj = ckpt_dir / "thresholds.json"
    if tj.exists():
        data = json.loads(tj.read_text(encoding="utf-8"))
        _AUTO_THRESHOLDS.update({k: float(v["auto"]) for k, v in data.items()})
    else:
        messages.append("⚠ 未找到 checkpoints/thresholds.json（自动判定阈值），"
                        "将按归一化热力图 0.5 兜底；可运行："
                        "python scripts/compute_auto_threshold.py")
    return messages


# ---------------------------------------------------------------------- #
# 预处理与推理
# ---------------------------------------------------------------------- #
def _preprocess(image: np.ndarray) -> torch.Tensor:
    """Gradio 图像 -> 归一化张量 (1,3,256,256)（PatchCore / CutPaste：cv2 双线性 resize，
    与训练/评估时的 MVTecDataset 预处理一致）。"""
    if image.shape[-1] == 4:  # RGBA -> RGB
        image = image[..., :3]
    if image.ndim == 2 or image.shape[-1] == 1:  # 灰度 -> RGB
        image = cv2.cvtColor(image, cv2.COLOR_GRAY2RGB)
    image = cv2.resize(image, (256, 256), interpolation=cv2.INTER_LINEAR)
    t = transforms.Compose([
        transforms.ToTensor(),
        transforms.Normalize(mean=IMAGENET_MEAN.tolist(), std=IMAGENET_STD.tolist()),
    ])(image)
    return t.unsqueeze(0)


# PaDiM 专用预处理：与 Anomalib 默认 PreProcessor 逐操作同源
# （uint8 -> float32[0,1] -> 抗锯齿 resize -> ImageNet 归一化；
#  Anomalib 评估管线为 read_image(float) + v2.Resize(antialias=True) + Normalize）
_PP_TV = T.Compose([
    T.ToImage(),
    T.ToDtype(torch.float32, scale=True),
    T.Resize((256, 256), antialias=True),
    T.Normalize(mean=IMAGENET_MEAN.tolist(), std=IMAGENET_STD.tolist()),
])


def _preprocess_tv(image: np.ndarray) -> torch.Tensor:
    """Gradio 图像 -> 归一化张量 (1,3,256,256)（PaDiM：float 上 torchvision 抗锯齿
    resize，与 Anomalib 评估管线逐像素一致）。"""
    if image.shape[-1] == 4:  # RGBA -> RGB
        image = image[..., :3]
    if image.ndim == 2 or image.shape[-1] == 1:  # 灰度 -> RGB
        image = cv2.cvtColor(image, cv2.COLOR_GRAY2RGB)
    t = _PP_TV(Image.fromarray(image))
    return t.unsqueeze(0)


# 模型类 -> 其评估管线使用的预处理（保证 Gradio 输出与报告评估同源）
_PREPROCESS = {
    PatchCoreModel: _preprocess,
    CutPasteModel: _preprocess,
    PadimAnomalibModel: _preprocess_tv,
}


def detect(image: np.ndarray, model_name: str, threshold_mult: float) -> tuple:
    """三模型推理入口（质检判定：异常得分 ≥ 判定阈值 → 不合格）。

    Args:
        threshold_mult: 灵敏度倍数，判定阈值 = threshold_mult × 自动阈值；
            调小更严格（检出率↑、误报率↑）、调大更宽松（反之）。

    Returns:
        (原图, 热力图叠加图, 缺陷掩码, 合格/不合格判定 HTML, 检测详情文本)
    """
    if image is None:
        raise gr.Error("请先上传一张图片")
    if model_name not in _MODELS:
        raise gr.Error(f"模型未加载，请先训练并保存权重。\n{load_models()}")

    model = _MODELS[model_name]
    preprocess = _PREPROCESS.get(type(model), _preprocess)
    x = preprocess(image).to(model.device)
    with torch.no_grad():
        raw_map = model.predict_image(x)
    score = float(raw_map.max())

    # 判定阈值 = 灵敏度倍数 × 自动阈值（未生成 thresholds.json 时按
    # 归一化热力图 0.5 兜底，即原始热力图值域中点）
    auto = _AUTO_THRESHOLDS.get(model_name)
    if auto is None:
        auto = (raw_map.max() - raw_map.min()) * 0.5 + raw_map.min()
    thresh = float(threshold_mult) * auto
    is_defect = score >= thresh

    # 可视化输出（热力图/掩码上采样回原始图像尺寸，保证叠加与原图对齐）
    original = image
    h, w = image.shape[:2]
    display_map = cv2.resize(raw_map, (w, h), interpolation=cv2.INTER_LINEAR)
    overlay = heatmap_overlay(image.astype(np.float32) / 255.0, display_map)
    # 缺陷掩码 = 异常得分 ≥ 判定阈值的区域（与判定结论同源）
    mask = (raw_map >= thresh).astype(np.uint8) * 255
    mask = cv2.resize(mask, (w, h), interpolation=cv2.INTER_NEAREST)

    verdict, color = ("不合格（检出缺陷）", "#dc2626") if is_defect else ("合格", "#16a34a")
    verdict_html = (
        f'<div style="font-size:30px;font-weight:bold;color:{color};'
        f'text-align:center;padding:14px;border:3px solid {color};'
        f'border-radius:10px;">{verdict}</div>'
    )
    auto_src = "自动阈值（训练集统计）" if model_name in _AUTO_THRESHOLDS else "兜底（归一化 0.5）"
    score_text = (f"异常得分：{score:.4f}\n"
                  f"判定阈值：{thresh:.4f} = {threshold_mult:.2f} × {auto:.4f}（{auto_src}）\n"
                  f"判定结果：{verdict}\n模型：{model_name}")
    return original, overlay, mask, verdict_html, score_text
```

### 6.2 文件：`scripts/compute_auto_threshold.py`（自动判定阈值（训练集统计 + 测试集验证））

```python
"""计算三模型的自动判定阈值，并在测试集上验证判定效果（质检判定，非评估协议）。

参考 B2 组螺丝缺陷检测系统的做法：随模型权重保存一个由训练集统计
得到的默认判定阈值，部署时「异常得分 ≥ 阈值」即判不合格（阈值可经
灵敏度倍数滑杆微调，默认 1.0 = 自动阈值）。

流程（每模型一次加载）：
1. 对 220 张正常训练样本逐一推理（走与 Gradio 部署完全相同的
   predict_image 路径），取异常得分最大值作为自动阈值（另记录
   p99/p99.5 供参考；若热力图整体落在概率尺度 [0,1] 且最大值饱和
   ≥0.9995，则以 p99 兜底，防止 GradCAM 类热力图在正常样本上饱和
   导致阈值失效）。PatchCore 例外：训练样本自身在记忆库中（自匹配
   距离≈0），改用留一法——取第二近邻距离（第一近邻是自身），与
   部署 k=1 得分同尺度；
2. 在测试集（115 张 = 22 正常 + 93 缺陷）上统计自动阈值下的判定
   表现：缺陷样本检出率、正常样本误报率。

输出：
- checkpoints/thresholds.json   随权重文件一起分发（Gradio 读取）
- results/threshold_verdict.json 判定验证结果（报告素材）

用法：
    PYTHONIOENCODING=utf-8 python scripts/compute_auto_threshold.py
"""

from __future__ import annotations

import json
import sys
from pathlib import Path

# 保证从项目根或 scripts/ 目录运行都能导入 src
sys.path.insert(0, str(Path(__file__).resolve().parent.parent))

import torch

from src import CHECKPOINT_DIR, RESULTS_DIR
from src.data import get_test_loader, get_train_loader
from src.experiment_utils import load_config, resolve
from src.models import CutPasteModel, PadimAnomalibModel, PatchCoreModel
from src.models.patchcore import VGG16_LAYERS, resize_map, smooth_map
from src.utils import set_seed

# Gradio 界面显示名 -> (模型类, 权重文件名)
MODEL_CLASSES = {
    "PatchCore（主方案）": (PatchCoreModel, "patchcore.pth"),
    "PaDiM": (PadimAnomalibModel, "padim.pth"),
    "CutPaste": (CutPasteModel, "cutpaste.pth"),
}


def _scores_on_loader(model, loader) -> list[float]:
    """对整个 loader 逐图推理，返回图像级异常得分列表（max 聚合，与部署一致）。"""
    scores = []
    with torch.no_grad():
        for batch in loader:
            x = batch["image"]
            for i in range(x.shape[0]):
                scores.append(float(model.predict_image(x[i : i + 1]).max()))
    return scores


def _patchcore_loo_score(model: PatchCoreModel, image: torch.Tensor) -> float:
    """PatchCore 留一法图像得分：每 patch 取第二近邻距离（第一近邻是训练样本
    自身，距离≈0），再做与部署一致的 resize+smooth，取图像最大值。

    尺度与部署 k=1 一致：测试图 k=1 的第一近邻与训练图 k=2 的第二近邻
    都是「库外最近真实匹配」距离。
    """
    feats_t = model._extract_patches_batch(image)  # (1,P,C)
    d2 = model._knn_gpu(feats_t.reshape(-1, feats_t.shape[-1]), 2)  # (P,2) 升序
    target_res = min(VGG16_LAYERS[n][2] for n in model.layers) // model.stride
    amap = d2[:, 1].reshape(target_res, target_res).cpu().numpy()
    amap = resize_map(amap, (model.image_size, model.image_size))
    return float(smooth_map(amap, sigma=4).max())


def _scores_on_loader_patchcore(model: PatchCoreModel, loader) -> list[float]:
    """PatchCore 训练集留一法得分列表（训练样本在库中，必须剔除自身匹配）。"""
    scores = []
    with torch.no_grad():
        for batch in loader:
            x = batch["image"]
            for i in range(x.shape[0]):
                scores.append(_patchcore_loo_score(model, x[i : i + 1]))
    return scores


def main() -> None:
    cfg = load_config()
    set_seed(cfg["dataset"]["seed"])
    data_root = resolve(cfg["paths"]["data_root"])
    device = "cuda" if torch.cuda.is_available() else "cpu"

    train_loader = get_train_loader(
        data_root,
        batch_size=cfg["dataset"]["batch_size"],
        num_workers=0,
        image_size=cfg["dataset"]["image_size"],
        use_augmentation=False,
    )
    test_loader = get_test_loader(
        data_root,
        batch_size=cfg["dataset"]["batch_size"],
        num_workers=0,
        image_size=cfg["dataset"]["image_size"],
    )

    thresholds: dict[str, dict] = {}
    verdicts: dict[str, dict] = {}
    for name, (cls, ckpt_name) in MODEL_CLASSES.items():
        model = cls(device=device)
        model.load(Path(CHECKPOINT_DIR) / ckpt_name)

        # 1) 训练集（仅正常）自动阈值；PatchCore 用留一法（剔除自身匹配）
        if isinstance(model, PatchCoreModel):
            scores = sorted(_scores_on_loader_patchcore(model, train_loader))
            max_note = "留一法（第二近邻）训练集正常样本异常得分最大值"
        else:
            scores = sorted(_scores_on_loader(model, train_loader))
            max_note = "训练集正常样本异常得分最大值"
        n = len(scores)
        auto = scores[-1]
        # 仅当热力图整体落在概率尺度 [0,1] 且最大值饱和时最大法才失效，
        # 改用 p99 兜底；马氏距离类尺度（PaDiM ≈ 10²）不适用此规则
        if scores[0] >= 0.0 and scores[-1] <= 1.0 + 1e-3 and auto >= 0.9995:
            auto = scores[int(n * 0.99) - 1]
            note = "最大值饱和，取 p99 兜底"
        else:
            note = max_note
        thresholds[name] = {
            "auto": round(auto, 6),
            "max": round(scores[-1], 6),
            "p99": round(scores[int(n * 0.99) - 1], 6),
            "p99.5": round(scores[int(n * 0.995) - 1], 6),
            "n_train": n,
            "note": note,
        }

        # 2) 测试集判定验证（label: 0=正常 1=缺陷）
        tp = fp = tn = fn = 0
        with torch.no_grad():
            for batch in test_loader:
                x, labels = batch["image"], batch["label"]
                for i in range(x.shape[0]):
                    flagged = float(model.predict_image(x[i : i + 1]).max()) >= auto
                    if labels[i].item() == 1:
                        tp += flagged; fn += not flagged
                    else:
                        fp += flagged; tn += not flagged
        recall = tp / (tp + fn)          # 缺陷检出率
        far = fp / (fp + tn)             # 正常误报率
        verdicts[name] = {
            "recall": round(recall, 4),
            "false_alarm": round(far, 4),
            "tp": tp, "fn": fn, "fp": fp, "tn": tn,
        }
        print(f"{name}: auto={auto:.4f} ({note})\n"
              f"    测试集判定：检出率 {recall:.2%}（{tp}/{tp + fn}）、"
              f"误报率 {far:.2%}（{fp}/{fp + tn}）")

    out = Path(CHECKPOINT_DIR) / "thresholds.json"
    out.write_text(json.dumps(thresholds, ensure_ascii=False, indent=2),
                   encoding="utf-8")
    vout = Path(RESULTS_DIR) / "threshold_verdict.json"
    vout.write_text(json.dumps(verdicts, ensure_ascii=False, indent=2),
                    encoding="utf-8")
    print(f"\n自动阈值已保存 -> {out}")
    print(f"判定验证结果已保存 -> {vout}")


if __name__ == "__main__":
    main()
```

### B.1 文件：`legacy/autoencoder.py`（AutoEncoder 重建基线（0.5293，已移除））

```python
"""卷积自编码器（重建基线，计划书模型 2）。

原理：仅用正常样本训练，编码器压缩为低维隐向量，解码器重建原图；
缺陷区域未被模型学习，重建残差大，残差大小即异常程度。

训练：Denoising MSE 损失（输入加高斯噪声，强迫解码器学习平滑先验，
对破坏纹理连续性的缺陷更敏感）+ Adam(1e-4) + Dropout + 早停（计划书 4.3）。
"""

from __future__ import annotations

import numpy as np
import torch
import torch.nn as nn

from .base import BaseAnomalyModel
from src.utils import smooth_map


class ConvAutoEncoder(nn.Module):
    """对称卷积自编码器（256×256 -> 8×8×256 隐空间 -> 256×256）。"""

    def __init__(self, latent_channels: int = 256, dropout: float = 0.2):
        super().__init__()
        # 编码器：每层 stride=2 下采样
        self.encoder = nn.Sequential(
            nn.Conv2d(3, 32, 4, 2, 1), nn.BatchNorm2d(32), nn.LeakyReLU(0.2),
            nn.Conv2d(32, 64, 4, 2, 1), nn.BatchNorm2d(64), nn.LeakyReLU(0.2),
            nn.Conv2d(64, 128, 4, 2, 1), nn.BatchNorm2d(128), nn.LeakyReLU(0.2),
            nn.Conv2d(128, 256, 4, 2, 1), nn.BatchNorm2d(256), nn.LeakyReLU(0.2),
            nn.Conv2d(256, latent_channels, 4, 2, 1), nn.BatchNorm2d(latent_channels),
            nn.LeakyReLU(0.2),
        )
        # Dropout 抑制过拟合（计划书：网络增加 Dropout 层）；
        # dropout=0 时退化为无正则基线（实验 4 基线实验用）
        self.dropout = nn.Dropout2d(dropout)
        # 解码器：转置卷积对称上采样
        self.decoder = nn.Sequential(
            nn.ConvTranspose2d(latent_channels, 256, 4, 2, 1), nn.BatchNorm2d(256), nn.ReLU(),
            nn.ConvTranspose2d(256, 128, 4, 2, 1), nn.BatchNorm2d(128), nn.ReLU(),
            nn.ConvTranspose2d(128, 64, 4, 2, 1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.ConvTranspose2d(64, 32, 4, 2, 1), nn.BatchNorm2d(32), nn.ReLU(),
            nn.ConvTranspose2d(32, 3, 4, 2, 1),
            nn.Tanh(),
        )

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        z = self.dropout(self.encoder(x))
        return self.decoder(z)


class AutoEncoderModel(BaseAnomalyModel):
    """AutoEncoder 异常检测模型（统一接口封装）。"""

    name = "autoencoder"

    def __init__(
        self,
        device="auto",
        image_size: int = 256,
        lr: float = 1e-4,
        epochs: int = 100,
        patience: int = 10,
        latent_channels: int = 256,
        noise: float = 0.05,  # Denoising AE 的输入噪声强度（仅训练时）
        dropout: float = 0.2,  # 隐空间 Dropout 概率（0 = 无正则基线）
    ):
        super().__init__(device, image_size)
        self.net = ConvAutoEncoder(latent_channels, dropout).to(self.device)
        self.lr, self.epochs, self.patience, self.noise = lr, epochs, patience, noise
        self.dropout = dropout

    def train(self, train_loader) -> None:
        """训练 + 早停：按 9:1 从训练集切出验证集监控 val loss。"""
        net = self.net.train()
        optimizer = torch.optim.Adam(net.parameters(), lr=self.lr)
        criterion = nn.MSELoss()

        # 从正常训练样本中切出验证集（仅用于早停，不参与梯度更新）
        all_images = []
        for batch in train_loader:
            all_images.append(batch["image"])
        images = torch.cat(all_images, dim=0)
        n_val = max(1, len(images) // 10)
        val_images = images[:n_val].to(self.device)
        train_images = images[n_val:]

        best_val, bad_epochs = float("inf"), 0
        n_train = len(train_images)
        print(f"[{self.name}] 训练集 {n_train} 张，验证集 {len(val_images)} 张，epochs={self.epochs}")
        for epoch in range(1, self.epochs + 1):
            perm = torch.randperm(n_train)
            total_loss = 0.0
            for i in range(0, n_train, 32):
                xb = train_images[perm[i : i + 32]].to(self.device)
                # Denoising：输入加高斯噪声，重建目标仍是干净图
                noisy = xb + torch.randn_like(xb) * self.noise
                rec = net(noisy)
                loss = criterion(rec, xb)
                optimizer.zero_grad()
                loss.backward()
                optimizer.step()
                total_loss += loss.item() * len(xb)
            train_loss = total_loss / n_train

            with torch.no_grad():
                val_loss = criterion(net(val_images), val_images).item()

            if val_loss < best_val - 1e-5:
                best_val, bad_epochs = val_loss, 0
                self._best_state = {k: v.clone() for k, v in net.state_dict().items()}
            else:
                bad_epochs += 1
            if epoch % 10 == 0 or bad_epochs == 0:
                print(f"  epoch {epoch:3d} | train loss {train_loss:.5f} | val loss {val_loss:.5f}")

            if bad_epochs >= self.patience:
                print(f"[{self.name}] 早停于 epoch {epoch}（{self.patience} 轮无改善）")
                break

        net.load_state_dict(self._best_state)
        net.eval()
        print(f"[{self.name}] 训练完成，最佳 val loss = {best_val:.5f}")

    @torch.no_grad()
    def predict_image(self, image: torch.Tensor) -> np.ndarray:
        """重建残差 -> 通道平均 -> 高斯平滑，得到异常热力图。"""
        self.net.eval()
        x = image.to(self.device)
        rec = self.net(x)
        residual = (x - rec).pow(2).mean(dim=1)  # (1,H,W) 逐像素残差
        amap = residual[0].cpu().numpy()
        # sigma=8：更大平滑核抑制单像素噪声，图像级得分更稳定
        return smooth_map(amap, sigma=8)

    def _state_dict(self) -> dict:
        return {"net": self.net.state_dict(), "config": {"lr": self.lr, "epochs": self.epochs}}

    def _load_state_dict(self, state: dict) -> None:
        self.net.load_state_dict(state["net"])
```

### B.2 文件：`legacy/deep_svdd.py`（DeepSVDD 深度单分类（0.4736，已移除））

```python
"""DeepSVDD 深度单分类模型（计划书模型 3）。

原理：将正常样本映射到特征空间紧凑超球体内；推理时计算样本特征到
球心距离，距离越大越可能异常。
- 防坍塌：球心 c 固定（AE 预训练隐码均值）+ 全连接层无偏置 + L2 正则
- 初始化：先训练自编码器，用其编码器权重初始化（计划书 4.3）
- 像素级定位：滑窗 Patch 逐一送入编码器，按到球心距离生成热力图
"""

from __future__ import annotations

import numpy as np
import torch
import torch.nn as nn

from .base import BaseAnomalyModel
from src.utils import resize_map, smooth_map


class SVDDEncoder(nn.Module):
    """编码器：全卷积 + 全局平均池化 + 无偏置全连接（输入尺寸无关）。"""

    def __init__(self, latent_dim: int = 128):
        super().__init__()
        self.convs = nn.Sequential(
            nn.Conv2d(3, 32, 4, 2, 1), nn.BatchNorm2d(32), nn.LeakyReLU(0.2),
            nn.Conv2d(32, 64, 4, 2, 1), nn.BatchNorm2d(64), nn.LeakyReLU(0.2),
            nn.Conv2d(64, 128, 4, 2, 1), nn.BatchNorm2d(128), nn.LeakyReLU(0.2),
            nn.Conv2d(128, 256, 4, 2, 1), nn.BatchNorm2d(256), nn.LeakyReLU(0.2),
            nn.Conv2d(256, 256, 4, 2, 1), nn.BatchNorm2d(256), nn.LeakyReLU(0.2),
        )
        self.gap = nn.AdaptiveAvgPool2d(1)
        self.fc = nn.Linear(256, latent_dim, bias=False)  # 无偏置，防止超球面坍塌

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        z = self.gap(self.convs(x)).flatten(1)
        return self.fc(z)


class SVDDDecoder(nn.Module):
    """解码器（仅 AE 预训练阶段使用）。"""

    def __init__(self, latent_dim: int = 128):
        super().__init__()
        self.fc = nn.Linear(latent_dim, 256 * 8 * 8)
        self.deconvs = nn.Sequential(
            nn.ConvTranspose2d(256, 256, 4, 2, 1), nn.BatchNorm2d(256), nn.ReLU(),
            nn.ConvTranspose2d(256, 128, 4, 2, 1), nn.BatchNorm2d(128), nn.ReLU(),
            nn.ConvTranspose2d(128, 64, 4, 2, 1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.ConvTranspose2d(64, 32, 4, 2, 1), nn.BatchNorm2d(32), nn.ReLU(),
            nn.ConvTranspose2d(32, 3, 4, 2, 1),
            nn.Tanh(),
        )

    def forward(self, z: torch.Tensor) -> torch.Tensor:
        return self.deconvs(self.fc(z).view(z.shape[0], 256, 8, 8))


class DeepSVDDModel(BaseAnomalyModel):
    """DeepSVDD：AE 预训练初始化 -> 固定球心 -> 距离最小化训练。"""

    name = "deep_svdd"

    def __init__(
        self,
        device="auto",
        image_size: int = 256,
        latent_dim: int = 128,
        lr: float = 1e-4,
        epochs: int = 100,
        ae_pretrain_epochs: int = 50,
        patch_size: int = 64,
        patch_stride: int = 16,
    ):
        super().__init__(device, image_size)
        self.latent_dim = latent_dim
        self.lr, self.epochs = lr, epochs
        self.ae_pretrain_epochs = ae_pretrain_epochs
        self.patch_size, self.patch_stride = patch_size, patch_stride
        self.encoder = SVDDEncoder(latent_dim).to(self.device)
        self.center: torch.Tensor | None = None  # 超球面球心

    # ------------------------------------------------------------------ #
    # 训练：AE 预训练 -> 固定球心 -> SVDD 距离最小化
    # ------------------------------------------------------------------ #
    def train(self, train_loader) -> None:
        # 1) 收集全部正常样本
        images = torch.cat([b["image"] for b in train_loader], dim=0)

        # 2) AE 预训练初始化编码器（计划书 4.3）；
        #    基线实验（实验 4）ae_pretrain_epochs=0 时跳过，退化为随机初始化
        if self.ae_pretrain_epochs > 0:
            self._pretrain_ae(images)

        # 3) 固定球心：正常样本隐码均值
        self.encoder.eval()
        with torch.no_grad():
            z = self.encoder(images.to(self.device))
        self.center = z.mean(dim=0, keepdim=True)  # (1,latent)

        # 4) SVDD 训练：最小化到球心距离
        self.encoder.train()
        optimizer = torch.optim.Adam(
            self.encoder.parameters(), lr=self.lr, weight_decay=1e-6
        )
        n = len(images)
        print(f"[{self.name}] SVDD 训练：{n} 张正常样本，epochs={self.epochs}")
        best_loss, best_state = float("inf"), None
        for epoch in range(1, self.epochs + 1):
            perm = torch.randperm(n)
            total = 0.0
            for i in range(0, n, 32):
                xb = images[perm[i : i + 32]].to(self.device)
                dist = (self.encoder(xb) - self.center).pow(2).sum(dim=1).mean()
                optimizer.zero_grad()
                dist.backward()
                optimizer.step()
                total += dist.item() * len(xb)
            avg = total / n
            if avg < best_loss:
                best_loss = avg
                best_state = {k: v.clone() for k, v in self.encoder.state_dict().items()}
            if epoch % 10 == 0:
                print(f"  epoch {epoch:3d} | 平均球心距离 {avg:.5f}")
        self.encoder.load_state_dict(best_state)
        self.encoder.eval()
        print(f"[{self.name}] 训练完成，最佳平均球心距离 = {best_loss:.5f}")

    def _pretrain_ae(self, images: torch.Tensor) -> None:
        """用自编码器预训练编码器权重。"""
        decoder = SVDDDecoder(self.latent_dim).to(self.device)
        params = list(self.encoder.parameters()) + list(decoder.parameters())
        optimizer = torch.optim.Adam(params, lr=1e-4)
        criterion = nn.MSELoss()
        n = len(images)
        print(f"[{self.name}] AE 预训练初始化（{self.ae_pretrain_epochs} epochs）...")
        self.encoder.train()
        decoder.train()
        for epoch in range(1, self.ae_pretrain_epochs + 1):
            perm = torch.randperm(n)
            total = 0.0
            for i in range(0, n, 32):
                xb = images[perm[i : i + 32]].to(self.device)
                rec = decoder(self.encoder(xb))
                loss = criterion(rec, xb)
                optimizer.zero_grad()
                loss.backward()
                optimizer.step()
                total += loss.item() * len(xb)
            if epoch % 10 == 0:
                print(f"  AE 预训练 epoch {epoch:3d} | MSE {total / n:.5f}")

    # ------------------------------------------------------------------ #
    # 推理：滑窗 Patch 到球心距离 -> 热力图
    # ------------------------------------------------------------------ #
    @torch.no_grad()
    def predict_image(self, image: torch.Tensor) -> np.ndarray:
        assert self.center is not None, "请先调用 train()"
        self.encoder.eval()
        x = image.to(self.device)  # (1,3,H,W)
        h, w = x.shape[-2], x.shape[-1]
        ps, st = self.patch_size, self.patch_stride

        # 滑窗裁剪全部 Patch，批量过编码器
        patches = []
        for i in range(0, h - ps + 1, st):
            for j in range(0, w - ps + 1, st):
                patches.append(x[:, :, i : i + ps, j : j + ps])
        grid_h = (h - ps) // st + 1
        grid_w = (w - ps) // st + 1
        patches_t = torch.cat(patches, dim=0)  # (P,3,ps,ps)
        z = self.encoder(patches_t)
        dist = (z - self.center).pow(2).sum(dim=1).cpu().numpy()  # (P,)
        amap = dist.reshape(grid_h, grid_w)
        amap = resize_map(amap, (h, w))
        return smooth_map(amap, sigma=4)

    # ------------------------------------------------------------------ #
    # 持久化
    # ------------------------------------------------------------------ #
    def _state_dict(self) -> dict:
        return {
            "encoder": self.encoder.state_dict(),
            "center": self.center,
            "latent_dim": self.latent_dim,
        }

    def _load_state_dict(self, state: dict) -> None:
        self.encoder.load_state_dict(state["encoder"])
        self.center = state["center"].to(self.device)
```
